-- ============================================
-- BASE DE DATOS: importadora_repuestos
-- ============================================
CREATE DATABASE IF NOT EXISTS importadora_repuestos;
USE importadora_repuestos;

-- ============================================
-- 1. TABLA: roles
-- ============================================
CREATE TABLE roles (
    id_rol INT PRIMARY KEY AUTO_INCREMENT,
    nombre_rol VARCHAR(50) NOT NULL UNIQUE,
    descripcion TEXT,
    nivel_permiso INT NOT NULL DEFAULT 1 COMMENT '1=mayor, 5=menor',
    estado TINYINT DEFAULT 1 COMMENT '1=activo, 0=inactivo',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ============================================
-- 2. TABLA: sucursales
-- ============================================
CREATE TABLE sucursales (
    id_sucursal INT PRIMARY KEY AUTO_INCREMENT,
    codigo_sucursal VARCHAR(20) NOT NULL UNIQUE,
    nombre_sucursal VARCHAR(100) NOT NULL,
    direccion TEXT,
    telefono VARCHAR(20),
    email VARCHAR(100),
    ciudad VARCHAR(50),
    estado TINYINT DEFAULT 1 COMMENT '1=activo, 0=inactivo',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ============================================
-- 3. TABLA: usuarios
-- ============================================
CREATE TABLE usuarios (
    id_usuario INT PRIMARY KEY AUTO_INCREMENT,
    id_rol INT NOT NULL,
    id_sucursal INT NULL COMMENT 'NULL para superadministrador',
    username VARCHAR(50) NOT NULL UNIQUE,
    email VARCHAR(100) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    intentos_fallidos INT DEFAULT 0,
    bloqueado_hasta DATETIME NULL,
    ultimo_login DATETIME NULL,
    ultima_ip VARCHAR(45),
    token_recuperacion VARCHAR(100) NULL,
    token_expira DATETIME NULL,
    estado TINYINT DEFAULT 1 COMMENT '1=activo, 0=inactivo, 2=bloqueado',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (id_rol) REFERENCES roles(id_rol),
    FOREIGN KEY (id_sucursal) REFERENCES sucursales(id_sucursal)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ============================================
-- 4. TABLA: usuarios_datos (información personal)
-- ============================================
CREATE TABLE usuarios_datos (
    id_usuario_dato INT PRIMARY KEY AUTO_INCREMENT,
    id_usuario INT NOT NULL,
    nombres VARCHAR(100) NOT NULL,
    apellidos VARCHAR(100) NOT NULL,
    cedula VARCHAR(20) UNIQUE,
    telefono_movil VARCHAR(20),
    telefono_fijo VARCHAR(20),
    direccion TEXT,
    fecha_nacimiento DATE,
    genero ENUM('M','F','OTRO') DEFAULT 'OTRO',
    foto_perfil VARCHAR(255),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (id_usuario) REFERENCES usuarios(id_usuario) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ============================================
-- 5. TABLA: logs_acceso (historial de intentos)
-- ============================================
CREATE TABLE logs_acceso (
    id_log INT PRIMARY KEY AUTO_INCREMENT,
    id_usuario INT NULL,
    email VARCHAR(100),
    ip_cliente VARCHAR(45),
    user_agent TEXT,
    tipo_evento ENUM('login_exitoso', 'login_fallido', 'bloqueo', 'logout', 'recuperacion') NOT NULL,
    descripcion TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_usuario (id_usuario),
    INDEX idx_email (email),
    INDEX idx_fecha (created_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ============================================
-- 6. TABLA: tokens_sesion (para manejo de sesiones)
-- ============================================
CREATE TABLE tokens_sesion (
    id_token INT PRIMARY KEY AUTO_INCREMENT,
    id_usuario INT NOT NULL,
    token VARCHAR(255) NOT NULL UNIQUE,
    ip_creacion VARCHAR(45),
    user_agent TEXT,
    fecha_expiracion DATETIME NOT NULL,
    activo TINYINT DEFAULT 1,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (id_usuario) REFERENCES usuarios(id_usuario) ON DELETE CASCADE,
    INDEX idx_token (token),
    INDEX idx_expiracion (fecha_expiracion)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- TABLA: configuracion_empresa

CREATE TABLE configuracion_empresa (
    id_config INT PRIMARY KEY AUTO_INCREMENT,
    -- Datos básicos
    razon_social VARCHAR(200) NOT NULL,
    nombre_comercial VARCHAR(200) NOT NULL,
    nit VARCHAR(20) NOT NULL UNIQUE,
    regimen VARCHAR(50) DEFAULT 'General',
    actividad_economica VARCHAR(100),
    
    -- Contacto
    telefono1 VARCHAR(20),
    telefono2 VARCHAR(20),
    email_principal VARCHAR(100),
    email_secundario VARCHAR(100),
    sitio_web VARCHAR(150),
    
    -- Ubicación
    pais VARCHAR(50) DEFAULT 'Bolivia',
    departamento VARCHAR(50),
    provincia VARCHAR(50),
    ciudad VARCHAR(50),
    zona VARCHAR(100),
    direccion TEXT,
    codigo_postal VARCHAR(20),
    
    -- Imágenes
    logo VARCHAR(255) DEFAULT 'default-logo.png',
    logo_favicon VARCHAR(255) DEFAULT 'default-favicon.ico',
    logo_blanco VARCHAR(255) DEFAULT 'default-logo-white.png',
    banner VARCHAR(255),
    
    -- Datos fiscales
    autorizacion_factura VARCHAR(50),
    numero_autorizacion VARCHAR(50),
    fecha_autorizacion DATE,
    llave_dosificacion VARCHAR(255),
    numero_sucursal VARCHAR(10),
    codigo_actividad VARCHAR(20),
    matricula_comercio VARCHAR(50),
    certificado_digital VARCHAR(255),
    
    -- Redes Sociales
    facebook VARCHAR(150),
    instagram VARCHAR(150),
    twitter VARCHAR(150),
    youtube VARCHAR(150),
    linkedin VARCHAR(150),
    tiktok VARCHAR(150),
    whatsapp VARCHAR(20),
    
    -- Configuración del sistema
    moneda_principal VARCHAR(10) DEFAULT 'BOB',
    moneda_secundaria VARCHAR(10),
    zona_horaria VARCHAR(50) DEFAULT 'America/La_Paz',
    formato_fecha VARCHAR(20) DEFAULT 'd/m/Y',
    formato_hora VARCHAR(20) DEFAULT 'H:i',
    decimales INT DEFAULT 2,
    separador_decimal VARCHAR(1) DEFAULT ',',
    separador_miles VARCHAR(1) DEFAULT '.',
    
    -- Metadatos
    estado TINYINT DEFAULT 1 COMMENT '1=Activo, 0=Inactivo',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    INDEX idx_nit (nit),
    INDEX idx_estado (estado)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
